7 · Tab strip (特性解释器)

default:
  rate: 100/min
  burst: 120
search.query:
  rate: 20/min
  burst: 40
  key: api_key
JS
document.querySelectorAll('.tab').forEach(b => b.onclick = () => {
  document.querySelectorAll('.tab').forEach(x => {
    x.style.borderBottomColor = 'transparent';
    x.style.color = 'var(--gray-500)';
  });
  b.style.borderBottomColor = 'var(--clay)';
  b.style.color = 'var(--slate)';
  document.querySelectorAll('[data-panel]').forEach(p => {
    p.hidden = p.dataset.panel !== b.dataset.tab;
  });
});

9 · Copy-to-clipboard button (编辑器导出)

JS
async function copyDemo(btn) {
  const text = '...whatever the user produced in the UI...';
  await navigator.clipboard.writeText(text);
  const old = btn.textContent;
  btn.textContent = 'Copied ✓';
  setTimeout(() => { btn.textContent = old; }, 1500);
}
